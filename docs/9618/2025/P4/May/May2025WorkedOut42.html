def Calculate():
      global Stack
      global TopOfStack
      Total = Pop()
      Total = int(Total)
      Return = 0
      LastOperator = ""
      Operator = True
      while(Return != "-1"):
          Return = Pop()
          if Operator == False:
              Data = int(Return)
              if LastOperator == "+":
                  Total = Total + Data
              elif LastOperator == "-":
                  Total = Total - Data
              elif LastOperator == "*":
                  Total = Total * Data
              elif LastOperator == "/":
                  Total = Total / Data
              elif LastOperator == "^":
                  Total = Total ** Data
              Operator = True
         else:
              LastOperator = Return
              Operator = False
     return Total
