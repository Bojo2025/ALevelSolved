Sub ReadHighScores()
                 Dim Textfile As String = "HighScore.txt"
                 Dim FileReader As New System.IO.StreamReader(textfile)

                   FileData(DataEntered, 0) = FileReader.ReadLine()
                   FileData(DataEntered, 1) = FileReader.ReadLine()
                   DataEntered = DataEntered + 1
                 End While
