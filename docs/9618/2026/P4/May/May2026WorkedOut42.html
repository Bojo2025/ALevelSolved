Function InsertionSort(TheData())
                      Dim ArrayLength As Integer = TheData.Length()
                      Dim Temp As String
                      Dim y As Integer
                      If ArrayLength <= 1 Then
                          Return TheData
                      End If
                      For x = 1 To ArrayLength - 1
                          Temp = TheData(x)
                          y = x - 1
                          While y >= 0
                              If Temp < TheData(y) Then
                                  TheData(y + 1) = TheData(y)
                                  y = y - 1
                              Else
                                  Exit While
                              End If
                          End While
                          TheData(y + 1) = Temp

def InsertionSort(TheData):
                       ArrayLength = len(TheData)
                       if ArrayLength <= 1:
                           return TheData
                       for x in range(1, ArrayLength):
                           Temp = TheData[x]
                           y = x-1
                           while y >= 0 and Temp < TheData[y]:
                                TheData[y+1] = TheData[y]
                                y -= 1
                           TheData[y+1] = Temp
                       return TheData
